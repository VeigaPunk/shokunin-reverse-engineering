#!/usr/bin/env python3
"""luna-round-8 lane-B pre-state census (seat XB1, census runner).

Read-only: Git objects of the pinned revision, on-disk receipts under
.ufo/waves/luna-round-8/receipts, and three host config files. No network,
no MCP, no tracked-file writes. Prints a deterministic report to stdout.
"""
import hashlib
import pathlib
import re
import subprocess
import sys

REV = '6e2ee8fadc4ab328ce7c8d4d82cceb9b326b94c4'
ROOT = pathlib.Path('/home/vgpnk/Projects/shokunin/kaidoku-shokunin')
RECEIPTS = ROOT / '.ufo/waves/luna-round-8/receipts'
SELF = 'census-laneB-pre.txt'

       'inspect_sqlite_database', 'search_bytes', 'inspect_with_objdump']

def run(cmd, **kw):
    return subprocess.run(cmd, capture_output=True, text=True, cwd=ROOT, **kw)


def git(*args):
    return subprocess.check_output(['git', *args], cwd=ROOT)


def blob(rev, path):
    return git('show', f'{rev}:{path}')


def tag(name, body):
    print(f'BEGIN-{name}')
    print(body, end='' if body.endswith('\n') or body == '' else '\n')
    print(f'END-{name}')


def main():
    out = []

    # [1] source controls
    head = git('rev-parse', 'HEAD').decode().strip()
    status = git('status', '--porcelain=v1').decode().splitlines()
    tracked_dirty = [l for l in status if not l.startswith('??')]
    diff_head = run(['git', 'diff', '--quiet', 'HEAD'])
    sc_worktree = hashlib.sha256((ROOT / 'SCORECARD.md').read_bytes()).hexdigest()
    sc_rev = hashlib.sha256(blob(REV, 'SCORECARD.md')).hexdigest()
    print(f'[1] SOURCE CONTROLS')
    print(f'PINNED_HEAD={REV}')
    print(f'OBSERVED_HEAD={head}')
    print(f'HEAD_MATCH={head == REV}')
    print(f'GIT_DIFF_QUIET_HEAD_EXIT={diff_head.returncode}')
    print(f'TRACKED_DIRTY_STATUS_LINES={len(tracked_dirty)}')
    print(f'UNTRACKED_STATUS_LINES={sum(l.startswith("??") for l in status)}')
    print(f'SCORECARD_WORKTREE_SHA256={sc_worktree}')
    print(f'SCORECARD_REV_BLOB_SHA256={sc_rev}')
    print(f'SCORECARD_WORKTREE_EQUALS_PINNED={sc_worktree == sc_rev}')
    print()

    # [2] Recipe A verbatim from the pinned SCORECARD blob
    sc = blob(REV, 'SCORECARD.md').decode()
    marker = sc.index('**A — source-derived controls.**')
    fence = re.search(r'```bash\n(.*?)\n```', sc[marker:], re.S)
    shell_recipe = fence.group(1)
    recipe = shell_recipe.split("<<'PY'\n", 1)[1].rsplit('\nPY', 1)[0]
    print(f'[2] RECIPE A VERBATIM (blob {REV[:7]}:SCORECARD.md, fence sha256 '
          f'{hashlib.sha256(shell_recipe.encode()).hexdigest()})')
    r = run([sys.executable, '-', REV], input=recipe)
    print(f'EXIT={r.returncode}')
    tag('STDOUT[recipeA]', r.stdout)
    tag('STDERR[recipeA]', r.stderr)
    print(f'STDOUT_SHA256={hashlib.sha256(r.stdout.encode()).hexdigest()}')
    print()

    # [3] coverage-map token census
    readme = blob(REV, 'corpus/README.md').decode()
    section = readme.split('## Coverage map\n', 1)[1].split('\n## ', 1)[0]
    tokens = re.findall(r'2026-10-\d{2}-[a-z0-9-]+', section)
    uniq = sorted(set(tokens))
    tok_sha = hashlib.sha256(('\n'.join(uniq) + '\n').encode()).hexdigest()
    paths = git('ls-tree', '-r', '--name-only', REV, '--', 'corpus/knowledge').decode().splitlines()
    stems = sorted(p.split('/')[-1][:-3] for p in paths if p.endswith('.md'))
    print('[3] COVERAGE-MAP TOKEN CENSUS (definition: Recipe A lines 101-102)')
    print(f'MAP_TOKEN_OCCURRENCES={len(tokens)}')
    print(f'MAP_TOKEN_UNIQUE={len(uniq)}')
    print(f'MAP_TOKEN_SHA256={tok_sha}')
    print(f'TOKENS_EQUAL_FILE_STEMS={uniq == stems}')
    print(f'MISSING={sorted(set(stems) - set(uniq))}')
    print(f'DANGLING={sorted(set(uniq) - set(stems))}')
    print()

    # [4] knowledge blob aggregate
    recs = []
    for s in stems:
        h = hashlib.sha256(blob(REV, f'corpus/knowledge/{s}.md')).hexdigest()
        recs.append(f'{s} {h}\n')
    agg = hashlib.sha256(''.join(recs).encode()).hexdigest()
    print('[4] KNOWLEDGE BLOB AGGREGATE')
    print(f'KNOWLEDGE_MD_COUNT={len(stems)}')
    print(f'KNOWLEDGE_MD_AGGREGATE_SHA256={agg}')
    print()

    # [5] receipt inventory + MANIFEST cross-check
    files = sorted(p for p in RECEIPTS.iterdir() if p.name != SELF)
    print('[5] RECEIPT INVENTORY (snapshot; excludes this census output file)')
    print(f'RECEIPT_FILE_COUNT={len(files)}')
    for p in files:
        if p.is_symlink():
            print(f'FILE {p.name}: symlink')
        elif p.is_file():
            data = p.read_bytes()
            print(f'FILE {p.name}: regular size={len(data)} sha256={hashlib.sha256(data).hexdigest()}')
        else:
            print(f'FILE {p.name}: non-regular')
    manifest = (RECEIPTS / 'MANIFEST.txt').read_text().splitlines()
    print('MANIFEST_CROSS-CHECK:')
    for line in manifest:
        name, claim = line.split()
        data = (RECEIPTS / name).read_bytes()
        actual = hashlib.sha256(data).hexdigest()
        if actual == claim:
            print(f'  {name}: MATCH')
            continue
        lines = data.splitlines(keepends=True)
        prefix = None
        for n in range(1, len(lines) + 1):
            if hashlib.sha256(b''.join(lines[:n])).hexdigest() == claim:
                prefix = n
                break
        if prefix is None:
            print(f'  {name}: MISMATCH actual={actual} (claimed {claim}; no line-prefix matches)')
        else:
            head_bytes = len(b''.join(lines[:prefix]))
            print(f'  {name}: APPEND-SUPERSEDED actual={actual} (claimed {claim} = head -n '
                  f'{prefix}, {head_bytes} B of {len(data)} B)')
    print()

    # [6] tool-list control (L2/L9 inventory-level)
    tools = [l for l in (RECEIPTS / 'leg-2-tools.txt').read_text().splitlines() if l.strip()]
    print('[6] 6.4.0 TOOL-LIST CONTROL (receipts/leg-2-tools.txt)')
    print(f'TOOL_LINE_COUNT={len(tools)}')
    print(f'TOOL_UNIQUE_COUNT={len(set(tools))}')
    print(f'TOOL_SORTED_SHA256={hashlib.sha256(("\n".join(sorted(set(tools))) + "\n").encode()).hexdigest()}')
    for name in SIX:
        r2 = subprocess.run(['grep', '-nFw', '--', name, 'leg-2-tools.txt'],
                            cwd=RECEIPTS, capture_output=True, text=True)
        print(f'ABSENCE {name}: rc={r2.returncode} stdout={r2.stdout.strip()!r}')
    print()

    # [7] config fingerprints
    print('[7] HOST-CONFIG FINGERPRINTS')
    for path in CONFIGS:
        if path.is_file():
            data = path.read_bytes()
            print(f'CONFIG {path}: regular size={len(data)} sha256={hashlib.sha256(data).hexdigest()}')
        else:
            print(f'CONFIG {path}: MISSING_OR_NONREGULAR')
    print()

    # [8] post-state + evidence snapshot
    head2 = git('rev-parse', 'HEAD').decode().strip()
    status2 = git('status', '--porcelain=v1').decode().splitlines()
    print('[8] POST-STATE')
    print(f'HEAD_AFTER={head2} UNCHANGED={head2 == head}')
    print(f'TRACKED_DIRTY_STATUS_LINES_AFTER={sum(not l.startswith("??") for l in status2)}')
    print(f'UNTRACKED_STATUS_LINES_AFTER={sum(l.startswith("??") for l in status2)}')
    wave = sorted(p.name for p in (ROOT / '.ufo/waves/luna-round-8').glob('*.md'))
    print(f'WAVE_REPORT_COUNT={len(wave)}')
    print('WAVE_REPORTS_EXIST_CB5=' + str((ROOT / '.ufo/waves/luna-round-8/CB5.md').exists()))
    return 0


if __name__ == '__main__':
    sys.exit(main())
