# ---------------------------------------------------------------- helpers ----


def run(*a, cwd=None):
    return subprocess.run(a, capture_output=True, text=True, cwd=cwd)


class CouldNotEvaluate(Exception):
    pass


def git_show(repo, rev, path):
    r = run("git", "show", f"{rev}:{path}", cwd=repo)
    if r.returncode != 0:
        raise CouldNotEvaluate(f"{rev}:{path} unreadable")
    return r.stdout


def read_graded(repo, at, path):
    """Bytes of `path` in the graded tree: a revision, or the working tree."""
    if at:
        return git_show(repo, at, path)
    real = os.path.join(repo, path)
    if not os.path.isfile(real):
        raise CouldNotEvaluate(f"{path} absent from the working tree")
    with open(real, "rb") as fh:
        return fh.read().decode("utf-8", errors="replace")


def rev_exists(repo, rev):
    return run("git", "rev-parse", "--verify", "--quiet",
               f"{rev}^{{commit}}", cwd=repo).returncode == 0


def diff_hunks(repo, pin, path, at):
    """Hunks of pin→graded for `path`. `at` None means the working tree."""
    args = ["git", "diff", "-U0", pin, "--", path]
    if at:
        args = ["git", "diff", "-U0", pin, at, "--", path]
    d = run(*args, cwd=repo).stdout
    out = []
    for line in d.splitlines():
        m = re.match(r"^@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@", line)
        if m:
            out.append(tuple(int(m.group(i) or 1) for i in (1, 2, 3, 4)))
    return out


def numstat(repo, pin, path, at):
    args = ["git", "diff", "--numstat", pin, "--", path]
    if at:
        args = ["git", "diff", "--numstat", pin, at, "--", path]
    return run(*args, cwd=repo).stdout.strip()

def check_pointer(repo, pin, at, p):
    """-> (True, None, detail) | (False, finding_id, detail) | (None, id, why)"""
    fid = p["id"]
    if p["kind"] == "symbol":
        real = os.path.expanduser(p["file"])
        if not os.path.isfile(real):
            return None, fid, f"symbol surface absent: {real}"
        lines = open(real, encoding="utf-8", errors="replace").read().splitlines()
        start = None
        for i, line in enumerate(lines, 1):
            if re.search(p["section"], line):
                start = i
                break
        if start is None:
            return False, fid, (f"section {p['section']!r} absent from {p['file']}")
        end = len(lines)
        for i in range(start, len(lines)):
            if lines[i].startswith("["):
                end = i
                break
        body = "\n".join(lines[start - 1:end])
        if re.search(p["must_match"], body):
            return True, None, (f"section {p['section']!r} at :{start} matches "
                                f"{p['must_match']!r} (line number NOT pinned)")
        return False, fid, (f"section {p['section']!r} at :{start} lacks "
                            f"{p['must_match']!r}")
    text = read_graded(repo, at, p["file"])
    lines = text.splitlines()
    n = p["line"]
    if n < 1 or n > len(lines):
        return False, fid, f"{p['file']}:{n} beyond EOF ({len(lines)} lines)"
    if re.search(p["must_match"], lines[n - 1]):
        return True, None, f"{p['file']}:{n} resolves"
    # drift locator: report where the pinned content actually went
    where = None
    try:
        pin_lines = git_show(repo, pin, p["file"]).splitlines()
        if 1 <= n <= len(pin_lines):
            needle = pin_lines[n - 1].strip()
            if needle:
                for i, line in enumerate(lines, 1):
                    if line.strip() == needle:
                        where = i
                        break
    except CouldNotEvaluate:
        pass
    moved = f"; identical line now at :{where}" if where else "; no identical line found"
    return False, fid, (f"{p['file']}:{n} lacks {p['must_match']!r}: "
                        f"{lines[n - 1][:60]!r}{moved}")


def check_append_only(repo, pin, at, a):
    fid = a["id"]
    path = a["path"]
    old = git_show(repo, pin, path)
    try:
        new = read_graded(repo, at, path)
    except CouldNotEvaluate:
        return True, None, f"{path} absent from the graded tree"
    if new.startswith(old):
        return True, None, f"{path} prefix-preserved (+{len(new) - len(old)} B)"
    old_lines, new_lines = old.splitlines(), new.splitlines()
    n = 0
    while (n < min(len(old_lines), len(new_lines))
           and old_lines[n] == new_lines[n]):
        n += 1
    return False, fid, (f"{path} edited below base length: first divergence at "
                        f"line {n + 1}")


def check_shape(repo, pin, at, s):
    fid = s["id"]
    path = s["path"]
    ns = numstat(repo, pin, path, at)
    if not ns:
        return True, None, f"{path} untouched"
    add, dele = (int(x) for x in ns.split("\t")[:2])
    hs = diff_hunks(repo, pin, path, at)
    oldlines = len(git_show(repo, pin, path).splitlines())
    bad = []
    for os_, oc, ns_, nc in hs:
        if os_ == 0 or oc == 0:
            if os_ == oldlines and s["append"]:
                continue
            if os_ in s["inserts"]:
                continue
            bad.append(f"insert/append hunk not allowed @@ -{os_},{oc}")
            continue
        lo, hi = os_, os_ + oc - 1
        if not any(lo >= a and hi <= z for a, z in s["allow"]):
            bad.append(f"hunk @@ -{os_},{oc} outside envelope {s['allow']}")
    if dele > s["max_del"]:
        bad.append(f"deletions {dele} > {s['max_del']}")
    if bad:
        return False, fid, f"{path}: " + "; ".join(bad)
    return True, None, f"{path}: +{add} -{dele} hunks={len(hs)}"


# ------------------------------------------------------------------ driver ---

def evaluate(repo, pin, at, contract, emit):
    findings, unknown = [], []
    for s in contract.get("snapshots", []):
        real = os.path.join(repo, s["path"])
        if not os.path.isfile(real):
            unknown.append(f"snapshot absent: {s['path']}")
            emit(f"SKIP {s['id']}: snapshot absent: {s['path']}")
            continue
        import hashlib
        got = hashlib.sha256(open(real, "rb").read()).hexdigest()
        if got != s["sha256"]:
            unknown.append(f"snapshot_tampered: {s['path']}")
            emit(f"SKIP {s['id']}: snapshot_tampered ({got[:12]}… != "
                 f"{s['sha256'][:12]}…)")
        else:
            emit(f"OK   {s['id']}: digest pinned ({got[:12]}…)")
    for p in contract["pointers"]:
        try:
            ok, fid, detail = check_pointer(repo, pin, at, p)
        except CouldNotEvaluate as e:
            unknown.append(str(e))
            emit(f"SKIP {p['id']}: {e}")
            continue
        if ok is None:
            unknown.append(detail)
            emit(f"SKIP {p['id']}: {detail}")
        elif ok:
            emit(f"OK   {p['id']}: {detail}")
        else:
            findings.append(fid)
            emit(f"FINDING {fid}  {detail}")
    for a in contract["append_only"]:
        try:
            ok, fid, detail = check_append_only(repo, pin, at, a)
        except CouldNotEvaluate as e:
            unknown.append(str(e))
            emit(f"SKIP {a['id']}: {e}")
            continue
        if ok:
            emit(f"OK   {a['id']}: {detail}")
        else:
            findings.append(fid)
            emit(f"FINDING {fid}  {detail}")
    for s in contract["shapes"]:
        try:
            ok, fid, detail = check_shape(repo, pin, at, s)
        except CouldNotEvaluate as e:
            unknown.append(str(e))
            emit(f"SKIP {s['id']}: {e}")
            continue
        if ok:
            emit(f"OK   {s['id']}: {detail}")
        else:
            findings.append(fid)
            emit(f"FINDING {fid}  {detail}")
    if findings:
        return 1, findings
    if unknown:
        return 2, findings
    return 0, findings

def selftest():
    """Isolated fixtures, fresh repo per case; asserts (exit, finding id)."""
    tmp = tempfile.mkdtemp(prefix="cdg-selftest-")
    env = dict(os.environ, GIT_AUTHOR_NAME="t", GIT_AUTHOR_EMAIL="t@t",
               GIT_COMMITTER_NAME="t", GIT_COMMITTER_EMAIL="t@t")

    def make(files):
        repo = tempfile.mkdtemp(dir=tmp)
        for name, body in files.items():
            real = os.path.join(repo, name)
            os.makedirs(os.path.dirname(real), exist_ok=True)
            with open(real, "w") as fh:
                fh.write(body)
        subprocess.run(["git", "init", "-q"], cwd=repo, env=env)
        subprocess.run(["git", "add", "-A"], cwd=repo, env=env)
        subprocess.run(["git", "commit", "-qm", "base"], cwd=repo, env=env)
        return repo, subprocess.run(["git", "rev-parse", "HEAD"], cwd=repo,
                                    capture_output=True, text=True,
                                    env=env).stdout.strip()

    def commit(repo, msg):
        subprocess.run(["git", "add", "-A"], cwd=repo, env=env)
        subprocess.run(["git", "commit", "-qm", msg], cwd=repo, env=env)

    cases = []

    # C1 clean: pointer resolves, append preserved, shape untouched
    repo, base = make({"SCORECARD.md": "one\n`:157` here\nthree\n",
                       "APPEND.md": "a\nb\nc\n"})
    cases.append((repo, base, {
        "pointers": [{"id": "p-ok", "kind": "line", "file": "SCORECARD.md",
                      "line": 2, "must_match": r":157"}],
        "append_only": [{"id": "a-ok", "path": "APPEND.md"}],
        "shapes": [{"id": "s-ok", "path": "SCORECARD.md", "allow": [],
                    "inserts": [], "append": True, "max_del": 0}]},
        (0, None)))

    # C2 dead pointer: cited line no longer carries the cited content
    repo, base = make({"SCORECARD.md": "one\n`:157` here\nthree\n"})

        "append_only": [], "shapes": []}, (1, "p-dead")))

    # C3 pointer past EOF
    repo, base = make({"SCORECARD.md": "one\n`:157` here\nthree\n"})
    cases.append((repo, base, {
        "pointers": [{"id": "p-eof", "kind": "line", "file": "SCORECARD.md",

        "append_only": [], "shapes": []}, (1, "p-eof")))

        fh.write("a\nB\nc\n")
    commit(repo, "edit-midfile")
    cases.append((repo, base, {
        "pointers": [], "shapes": [],
        "append_only": [{"id": "a-broke", "path": "APPEND.md"}]},

    # C5 append-only clean: pure EOF append keeps the prefix
    repo, base = make({"APPEND.md": "a\nb\nc\n"})

                      "section": r"^\[mcp_servers\.rea\]",

        "append_only": [], "shapes": []}, (2, "sym-absent")))

    tally, bad = [], 0
    for i, (repo, base, contract, (want_code, want_fid)) in enumerate(cases, 1):
        out = []
        code, findings = evaluate(repo, base, None, contract, out.append)
        # A could-not-evaluate finding surfaces as a SKIP line, not a FINDING.
        seen = findings + [ln.split(":", 1)[0].split()[-1]
                           for ln in out if ln.startswith("SKIP ")]
        ok = code == want_code and (want_fid is None or want_fid in seen)
        if not ok:
            bad += 1
        tally.append(f"C{i} {'ok  ' if ok else 'FAIL'} exit={code} "
                     f"want={want_code} findings={findings or '[]'}")
    print("\n".join(tally))
    print(f"selftest {'PASS' if bad == 0 else 'FAIL'} "

    return 0 if bad == 0 else 1


# -------------------------------------------------------------------- main ---

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--repo", default=".")
    ap.add_argument("--pin", default=None)
    ap.add_argument("--at", default=None)
    ap.add_argument("--contract", default=None)
    ap.add_argument("--json", action="store_true")
    ap.add_argument("--selftest", action="store_true")
    args = ap.parse_args()

    if args.selftest:
        return selftest()

    if args.contract:
        try:
            with open(args.contract, encoding="utf-8") as fh:
                contract = json.load(fh)
        except Exception as e:
            print(f"VERDICT could-not-evaluate\ncontract unreadable: {e}")
            return 2
    else:
        contract = DEFAULT_CONTRACT
    if args.pin:
        contract = dict(contract, pin_rev=args.pin)

    if not rev_exists(args.repo, contract["pin_rev"]):
        print(f"VERDICT could-not-evaluate\npin revision "
              f"{contract['pin_rev']} absent from {args.repo}")
        return 2

    lines = []
    code, findings = evaluate(args.repo, contract["pin_rev"], args.at,
                              contract, lines.append)
    verdict = {0: "clean", 1: "failed", 2: "could-not-evaluate"}[code]
    if args.json:
        print(json.dumps({"verdict": verdict, "findings": findings,
                          "pin_rev": contract["pin_rev"],
                          "graded": args.at or "worktree"}, indent=2))
        return code
    print("\n".join(lines))
    print(f"VERDICT {verdict}")
    return code


if __name__ == "__main__":
    sys.exit(main())
