```bash
#!/usr/bin/env bash
# R8 (round 3) marker-vs-report checker v3. Usage: check-report-v3.sh <repo> <report-path> [<expected-round-work-hash>] [<wave-dir>]
# Exit 0 = real report (maybe defective); 1 = marker; 2 = could-not-evaluate (missing file / usage).
# v3 vs v2: (a) a missing report file exits 2 (could-not-evaluate), not 1 (marker);
#           (b) P6b flags tmpfs/session-scoped pointers (/tmp, /dev/shm) as contract defects;
#           (c) marker message names the conjunctive bar explicitly (the bar itself is
#               unchanged from v2: no resolvable commit AND no resolvable pointer).
set -u
repo=$1; rpt=$2; work=${3:-}; wave=${4:-}
cd "$repo" || exit 2
if [ ! -f "$rpt" ]; then echo "P1 path-exists: FAIL (could-not-evaluate: no report at path, PROTOCOL.md:154-157)"; exit 2; fi
echo "P1 path-exists: OK"

grep -qE '^#{1,6}[[:space:]].*SHONEN' "$rpt" && echo "P2 shonen-heading: OK" || echo "P2 shonen-heading: FAIL (PROTOCOL.md:148-150)"

marks=$(awk '/^#{1,6}[[:space:]].*SHONEN/{f=1;next} f&&/^- \[[xX~ ]\]/{n++} END{print n+0}' "$rpt")
echo "P3 shonen-marks: $marks (expect 4)"

commits=0; commitok=0; dig=0; other=0
for h in $(grep -oE '\b[0-9a-f]{7,40}\b' "$rpt" | sort -u); do
  if git cat-file -e "$h^{commit}" 2>/dev/null; then
    commits=$((commits+1))
    if git merge-base --is-ancestor "$h" HEAD 2>/dev/null; then commitok=$((commitok+1)); else echo "P4 WARN non-ancestor commit $h"; fi
  elif grep -qE "$h…|$h\.\.\." "$rpt"; then dig=$((dig+1))
  elif git cat-file -e "$h" 2>/dev/null; then other=$((other+1))
  else echo "P4 NOTE unresolvable hex token $h (not a commit, not a marked digest)"; fi
done
echo "P4 commits: $commitok/$commits resolvable+ancestor; digest-prefixes: $dig (not commits); blobs: $other"
if [ -n "$work" ]; then
  git cat-file -e "$work^{commit}" 2>/dev/null && git merge-base --is-ancestor "$work" HEAD 2>/dev/null \
    && echo "P4b ancestor($work): OK" || echo "P4b ancestor($work): FAIL"
fi

c=$(git log -1 --format='%H %an <%ae>' -- "$rpt" 2>/dev/null)
[ -n "$c" ] && echo "P5 report-committed: OK [$c]" || echo "P5 report-committed: FAIL (untracked; lands-last unsatisfied PROTOCOL.md:122)"

n=0; ok=0; skip=0
while IFS= read -r tok; do
  case "$tok" in *\ *|*'<'*|*'$'*|*'|'*) skip=$((skip+1)); continue;; esac
  case "$tok" in */*|*.md|*.json|*.txt) ;; *) continue;; esac
  n=$((n+1)); p="${tok%%:*}"
  if [ -e "$p" ] || [ -e "$tok" ]; then ok=$((ok+1));
  elif [ -n "$wave" ] && [ -e "$wave/$p" ]; then ok=$((ok+1));
  else echo "P6 MISS $tok"; fi
done < <(grep -oE '`[^`]+`' "$rpt" | tr -d '`')
echo "P6 pointers-resolvable: $ok/$n (skipped $skip shell/placeholder tokens)"

tmpn=$(grep -oE '`(/tmp|/dev/shm)[^`]*`' "$rpt" | sort -u | wc -l)
if [ "$tmpn" -gt 0 ]; then
  echo "P6b tmpfs pointers: $tmpn (contract defect, PROTOCOL.md:341-345)"
  grep -oE '`(/tmp|/dev/shm)[^`]*`' "$rpt" | sort -u | sed 's/^/     /'
else echo "P6b tmpfs pointers: 0"; fi

sec=0
for s in 'Work performed' 'Landed changes' 'Before.*after' 'SHONEN'; do grep -qiE "^#{1,6}[[:space:]].*$s" "$rpt" && sec=$((sec+1)); done
echo "P7 sections(work/landed/before-after/shonen): $sec/4"

stale=$(grep -ciE 'stale-lane|two driver ticks|progressing|silent' "$rpt")
echo "P8 stale-lane duty line: $stale hit(s) (PROTOCOL.md:158-163)"

if [ "$commitok" -eq 0 ] && [ "$ok" -eq 0 ] && [ "$sec" -eq 0 ]; then
  echo "VERDICT: MARKER (no resolvable commit, no resolvable pointer, no sections -> integrator records failed, PROTOCOL.md:184-189)"; exit 1
elif [ "$commitok" -eq 0 ] && [ "$ok" -eq 0 ]; then
  echo "VERDICT: MARKER (no resolvable commit and no resolvable pointer -> failed, PROTOCOL.md:184-189)"; exit 1
elif [ "$marks" -lt 4 ] || [ "$sec" -lt 4 ]; then
  echo "VERDICT: DEFECTIVE REAL REPORT (names real work; minimum-content defect recorded, PROTOCOL.md:188-189)"; exit 0
else
  echo "VERDICT: REAL REPORT"; exit 0
fi
```
