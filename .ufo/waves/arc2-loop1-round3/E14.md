### 5a — `scripts/citation-drift.py` (sha256 `2e4f30e5…`, 409 lines)

```python
#!/usr/bin/env python3
"""citation-drift gate v1 (kaidoku-shokunin) — needle-at-line check over a pinned pointer table.

A committed, re-runnable verdict script per `PROTOCOL.md` §Gate standard: it reads
the pinned pointer table (`scripts/citation-pointers.tsv`) and asserts, for every
row, that the needle text is present on the cited line of the cited file. The
table's own sha256 is pinned in `scripts/citation-pointers.tsv.sha256`, so a
hand-edit that deletes a failing row is `table_tampered` (could-not-evaluate),
not a silent pass.

Table format (TSV, one row per line; blank lines and `#` comments ignored):
    <path>\t<line>\t<needle>\t<scope>
  path    repo-relative, absolute, or `~`-prefixed (host rows)
  line    1-based line number
  needle  substring that must appear on that line, matched against the
          whitespace-stripped line, so re-indentation does not break a pin
  scope   `repo`  — gated: a drifted/missing row is the defect this gate owns
          `host`  — advisory: the cited surface is operator-owned and outside
                    lane control (PROTOCOL §Drift pins, "RECOMMENDED"); host
                    rows are reported and never change the exit code

Verdict map (also declared in the SKILL.md registry row):
  0  clean               every repo row resolves (host rows may drift: advisory)
  1  failed              >=1 repo row is drifted / missing — the needle is
                         absent at the cited line; `drifted` reports the new
                         line number when the needle occurs exactly once
  2  could-not-evaluate  no repo row failed, but >=1 repo row is ambiguous,
                         target_missing or target_unreadable, or the table is
                         missing / malformed / tampered

Frozen evidence (Gate standard, "Two evidences"): `--snapshot DIR` copies the
table and every repo-scope target into DIR, writes DIR/manifest.json with a
sha256 per file plus the verdict computed at snapshot time, and refuses to
snapshot a non-clean verdict. `--frozen` replays the snapshot, verifies the
digests first (mismatch = `snapshot_tampered`, could-not-evaluate) and then
compares the recomputed verdict against the pinned one (`frozen_verdict_mismatch`
= failed). Host-scope files are never copied into the repo by `--snapshot`.

Output never echoes target-file text: only the lane-authored needle, the
path:line and the class, so the gate's stdout is safe to paste into reports.

Write scope: normal, `--frozen` and `--selftest` entries are read-only over the
repo; `--snapshot DIR` is the only writer and writes only under DIR; the selftest
creates and removes only its own temp directory under the system temp.

Usage:
  python3 scripts/citation-drift.py [--root DIR] [--table REL] [--json]
  python3 scripts/citation-drift.py --snapshot scripts/citation-drift-frozen
  python3 scripts/citation-drift.py --frozen
  python3 scripts/citation-drift.py --selftest
Re-pin the table after an intentional edit:
  sha256sum scripts/citation-pointers.tsv > scripts/citation-pointers.tsv.sha256
"""

import argparse
import hashlib
import json
import os
import re
import subprocess
import sys
import tempfile
from pathlib import Path

TABLE_NAME = "citation-pointers.tsv"
TABLE_REL = f"scripts/{TABLE_NAME}"
TABLE_SHA_REL = f"{TABLE_REL}.sha256"
FROZEN_REL = "scripts/citation-drift-frozen"
SCOPES = ("repo", "host")
FAILING = ("drifted", "missing")
COULD_NOT_EVALUATE = ("ambiguous", "target_missing", "target_unreadable")


class Finding(Exception):
    def __init__(self, code, name, detail=""):
        super().__init__(name)
        self.code, self.name, self.detail = code, name, detail


def sha256_bytes(data):
    return hashlib.sha256(data).hexdigest()


def sha256_file(path):
    return sha256_bytes(Path(path).read_bytes())


def revision(root):
    try:
        head = subprocess.run(["git", "-C", str(root), "rev-parse", "HEAD"],
                              capture_output=True, text=True)
        if head.returncode != 0:
            return "unknown"
        status = subprocess.run(["git", "-C", str(root), "status", "--porcelain"],
                                capture_output=True, text=True)
        return head.stdout.strip() + ("+dirty" if status.stdout.strip() else "")
    except OSError:
        return "unknown"


def parse_table(text):
    rows = []
    for number, raw in enumerate(text.splitlines(), 1):
        if not raw.strip() or raw.strip().startswith("#"):
            continue
        fields = raw.split("\t")
        if len(fields) != 4:
            raise Finding(2, "table_malformed",
                          f"row {number}: expected 4 tab-separated fields, got {len(fields)}")
        path, line_text, needle, scope = (field.strip() for field in fields)
        if not path or not needle or scope not in SCOPES:
            raise Finding(2, "table_malformed",
                          f"row {number}: empty path/needle or scope {scope!r}")
        if not re.fullmatch(r"\d+", line_text) or int(line_text) < 1:
            raise Finding(2, "table_malformed", f"row {number}: bad line {line_text!r}")
        rows.append({"row": number, "path": path, "line": int(line_text),
                     "needle": needle, "scope": scope})
    if not rows:
        raise Finding(2, "table_malformed", "table has no rows")
    return rows


def target_path(root, raw):
    if raw.startswith("~"):
        return Path(os.path.expanduser(raw))
    path = Path(raw)
    return path if path.is_absolute() else Path(root) / path


def classify(root, row):
    out = dict(row)
    try:
        text = target_path(root, row["path"]).read_text(encoding="utf-8")
    except FileNotFoundError:
        out["class"] = "target_missing"
        return out
    except (OSError, UnicodeDecodeError) as error:
        out["class"] = "target_unreadable"
        out["detail"] = type(error).__name__
        return out
    lines = text.splitlines()
    index = row["line"] - 1
    if 0 <= index < len(lines) and row["needle"] in lines[index].strip():
        out["class"] = "resolved"
        return out
    hits = [number for number, line in enumerate(lines, 1) if row["needle"] in line.strip()]
    if not hits:
        out["class"] = "missing"
        out["detail"] = "line_out_of_range" if index >= len(lines) else "needle_absent"
        return out
    if len(hits) == 1:
        out["class"] = "drifted"
        out["found_at"] = hits[0]
        return out
    out["class"] = "ambiguous"
    out["found_at"] = hits[:8]
    return out


def evaluate(root, rows):
    classified = [classify(root, row) for row in rows]
    repo = [row for row in classified if row["scope"] == "repo"]
    host = [row for row in classified if row["scope"] == "host"]
    failed = [row for row in repo if row["class"] in FAILING]
    could_not = [row for row in repo if row["class"] in COULD_NOT_EVALUATE]
    advisory = [row for row in host if row["class"] != "resolved"]
    code = 1 if failed else (2 if could_not else 0)
    first = (failed + could_not)[0]["class"] if (failed or could_not) else "clean"
    return {"exit": code, "first_finding": first, "rows": classified,
            "counts": {"rows": len(classified), "repo": len(repo), "host": len(host),
                       "failed": len(failed), "could_not_evaluate": len(could_not),
                       "advisory": len(advisory)}}


def identity(result):
    return {"exit": result["exit"],
            "counts": {key: result["counts"][key] for key in ("repo", "failed", "could_not_evaluate")},
            "rows": [[row["path"], row["line"], row["scope"], row["class"], row.get("found_at")]
                     for row in result["rows"]]}


def render(result, revision_text):
    lines = []
    for row in result["rows"]:
        found = row.get("found_at")
        if isinstance(found, list):
            found = ",".join(str(value) for value in found)
        elif found is None:
            found = "-"
        detail = f" detail={row['detail']}" if row.get("detail") else ""
        lines.append(f'CITATION_DRIFT_ROW: {row["class"]} {row["scope"]} '
                     f'{row["path"]}:{row["line"]} found_at={found} '
                     f'needle="{row["needle"]}"{detail}')
    state = {0: "CLEAN", 1: "FAILED", 2: "COULD_NOT_EVALUATE"}[result["exit"]]
    counts = result["counts"]
    lines.append(f'CITATION_DRIFT: {state} finding={result["first_finding"]} '
                 f'rows={counts["rows"]} repo={counts["repo"]} host={counts["host"]} '
                 f'failed={counts["failed"]} could_not_evaluate={counts["could_not_evaluate"]} '
                 f'advisory={counts["advisory"]} revision={revision_text}')
    return lines


def load_checked_table(root, table_rel):
    table = Path(root) / table_rel
    if not table.is_file():
        raise Finding(2, "table_missing", str(table))
    digest_file = Path(root) / (table_rel + ".sha256")
    if not digest_file.is_file():
        raise Finding(2, "table_sha_missing", str(digest_file))
    match = re.search(r"\b[0-9a-f]{64}\b", digest_file.read_text(encoding="utf-8", errors="replace"))
    if not match:
        raise Finding(2, "table_sha_missing", f"no digest in {digest_file}")
    actual = sha256_file(table)
    if actual != match.group(0):
        raise Finding(2, "table_tampered",
                      f"{table_rel} sha256 {actual[:16]}… != pinned {match.group(0)[:16]}…")
    return table, parse_table(table.read_text(encoding="utf-8"))


def take_snapshot(root, table_rel, rows, out_dir):
    root = Path(root).resolve()
    out_dir = Path(out_dir)
    repo_rows = [row for row in rows if row["scope"] == "repo"]
    result = evaluate(root, repo_rows)
    if result["exit"] != 0:
        raise Finding(2, "snapshot_not_clean",
                      f"live repo verdict is {result['first_finding']}; fix or re-pin before freezing")
    out_dir.mkdir(parents=True, exist_ok=True)
    files = {}
    for row in repo_rows:
        source = target_path(root, row["path"]).resolve()
        try:
            source.relative_to(root)
        except ValueError:
            raise Finding(2, "snapshot_refused",
                          f"row {row['row']} target outside root: {row['path']}")
        destination = out_dir / "files" / row["path"]
        destination.parent.mkdir(parents=True, exist_ok=True)
        destination.write_bytes(source.read_bytes())
        files[row["path"]] = sha256_file(source)
    table = Path(root) / table_rel
    (out_dir / TABLE_NAME).write_bytes(table.read_bytes())
    manifest = {"table": TABLE_NAME, "table_sha256": sha256_file(table),
                "files": files, "measured_revision": revision(root),
                "expected_repo_identity": identity(result)}
    (out_dir / "manifest.json").write_text(json.dumps(manifest, indent=1, sort_keys=True) + "\n")
    return manifest


def run_frozen(root, frozen_rel):
    directory = Path(root) / frozen_rel
    manifest_path = directory / "manifest.json"
    if not manifest_path.is_file():
        raise Finding(2, "frozen_missing", str(manifest_path))
    try:
        manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
        files = manifest["files"]
        expected = manifest["expected_repo_identity"]
    except (ValueError, KeyError, TypeError) as error:
        raise Finding(2, "frozen_missing", f"manifest unreadable: {type(error).__name__}")
    if not files or not all(isinstance(digest, str) and len(digest) == 64 for digest in files.values()):
        raise Finding(2, "frozen_unpinned",
                      "manifest pins no per-file sha256 (unpinned legacy baseline)")
    for relative, digest in sorted(files.items()):
        snapshot = directory / "files" / relative
        if not snapshot.is_file():
            raise Finding(2, "snapshot_tampered", f"missing {relative}")
        if sha256_file(snapshot) != digest:
            raise Finding(2, "snapshot_tampered", relative)
    table = directory / manifest["table"]
    if not table.is_file():
        raise Finding(2, "frozen_table_missing", manifest["table"])
    if sha256_file(table) != manifest["table_sha256"]:
        raise Finding(2, "snapshot_tampered", "frozen table")
    rows = [row for row in parse_table(table.read_text(encoding="utf-8")) if row["scope"] == "repo"]
    result = evaluate(directory / "files", rows)
    if identity(result) != expected:
        raise Finding(1, "frozen_verdict_mismatch",
                      f"recomputed {identity(result)['counts']} != pinned {expected['counts']}")
    print(f"CITATION_DRIFT_FROZEN: CLEAN finding=reproduces rows={len(rows)} "
          f"verdict_exit={expected['exit']} measured_revision={manifest.get('measured_revision')}")
    return 0


def build_fixture(base, table_text, files, digest=None):
    (base / "scripts").mkdir(parents=True, exist_ok=True)
    table = base / "scripts" / TABLE_NAME
    table.write_text(table_text, encoding="utf-8")
    for relative, text in files.items():
        target = base / relative
        target.parent.mkdir(parents=True, exist_ok=True)
        target.write_text(text, encoding="utf-8")
    pinned = digest if digest is not None else sha256_file(table)
    (base / "scripts" / (TABLE_NAME + ".sha256")).write_text(
        f"{pinned}  scripts/{TABLE_NAME}\n", encoding="utf-8")
```
