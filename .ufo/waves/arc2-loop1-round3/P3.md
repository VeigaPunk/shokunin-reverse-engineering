### 7.A `verify-citations.py` (sha256 `c043b571…`, 251 lines)

```python
#!/usr/bin/env python3
"""citation-drift gate v1 (draft, seat P3, wave arc2-loop1-round3).

Verdict script over pinned `file:line` citations. Reads a committed pointer
table (pointers.json) and asserts, per row, that the citing line still carries
the citation and that the cited line still carries the cited content (needle).

Exit map (declared in the SKILL.md registry row):
  0 clean             every row evaluated and resolved
  1 failed            >=1 evaluated row drifted (positive finding outranks
                      undecidability)
  2 could-not-evaluate table unreadable/malformed, a repo-side input absent,
                      or >=1 external row whose guard digest no longer matches
Any other code is undeclared and reads as could-not-evaluate.

Row kinds:
  pin   — source file:line must carry source_needle; target file:line must
          carry target_needle. Both sides line-pinned.
  locus — source_needle must appear somewhere in the source file (survives
          append-only errata landing at an unpinned line); target file:line
          must still carry target_needle.

External surfaces (host configs/shell files, outside the lane's control):
a row carrying `external: {path, sha256}` is evaluated only while the live
file's digest equals the pinned digest (the round's committed pre-pair value).
A moved external surface reports `external_moved` for that row and makes the
run could-not-evaluate (2) unless a drift was already found (then 1).
"""
import argparse
import hashlib
import json
import subprocess
import sys
import tempfile
from pathlib import Path

HERE = Path(__file__).resolve().parent


class Finding(Exception):
    def __init__(self, code, name):
        self.code, self.name = code, name


def load_table(path):
    try:
        table = json.loads(path.read_text())
    except (OSError, ValueError):
        raise Finding(2, "table_malformed")
    if not isinstance(table, dict) or not isinstance(table.get("rows"), list):
        raise Finding(2, "table_malformed")
    for entry in table.get("scoped_out", []):
        if not isinstance(entry, dict) or set(entry) != {"pointer", "file", "reason"}:
            raise Finding(2, "table_malformed")
    for row in table["rows"]:
        for key in ("id", "kind", "source", "source_needle", "target",
                    "target_line", "target_needle"):
            if key not in row:
                raise Finding(2, "table_malformed")
        if row["kind"] not in ("pin", "locus"):
            raise Finding(2, "table_malformed")
        if row["kind"] == "pin" and "source_line" not in row:
            raise Finding(2, "table_malformed")
        if "external" in row:
            if set(row["external"]) != {"path", "sha256"}:
                raise Finding(2, "table_malformed")
    return table


def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def read_lines(root, path):
    file = Path(path)
    if not file.is_absolute():
        file = root / path
    try:
        return file.read_text().splitlines()
    except OSError:
        raise Finding(2, "could_not_evaluate:" + path)


def line_at(lines, number):
    return lines[number - 1] if 1 <= number <= len(lines) else None


def check(root, table):
    drifted, unevaluated = [], []
    for row in table["rows"]:
        if "external" in row:
            live = row["external"]["path"]
            try:
                live_digest = digest(live)
            except OSError:
                unevaluated.append((row["id"], "external_absent", live))
                continue
            if live_digest != row["external"]["sha256"]:
                unevaluated.append((row["id"], "external_moved", live))
                continue
        source = read_lines(root, row["source"])
        target = read_lines(root, row["target"])
        if row["kind"] == "pin":
            line = line_at(source, row["source_line"])
            if line is None or row["source_needle"] not in line:
                drifted.append((row["id"], "source_line",
                                f"{row['source']}:{row['source_line']}",
                                row["source_needle"]))
                continue
        else:
            if not any(row["source_needle"] in line for line in source):
                drifted.append((row["id"], "source_locus", row["source"],
                                row["source_needle"]))
                continue
        line = line_at(target, row["target_line"])
        if line is None or row["target_needle"] not in line:
            drifted.append((row["id"], "target_line",
                            f"{row['target']}:{row['target_line']}",
                            row["target_needle"]))
    return drifted, unevaluated


def report(root, table):
    drifted, unevaluated = check(root, table)
    for entry in table.get("scoped_out", []):
        print(f"CITATION_SCOPED_OUT pointer={entry['pointer']} "
              f"file={entry['file']} reason={entry['reason']}")
    for row_id, side, where, needle in drifted:
        print(f"CITATION_DRIFT row={row_id} side={side} at={where} "
              f"expected_needle={needle!r}")
    for row_id, side, where in unevaluated:
        print(f"CITATION_UNEVALUATED row={row_id} side={side} at={where}")
    if drifted:
        print(f"CITATION-DRIFT: FAILED drifted={len(drifted)} "
              f"unevaluated={len(unevaluated)} rows={len(table['rows'])}")
        return 1
    if unevaluated:
        print(f"CITATION-DRIFT: COULD_NOT_EVALUATE "
              f"unevaluated={len(unevaluated)} rows={len(table['rows'])}")
        return 2
    print(f"CITATION-DRIFT: CLEAN rows={len(table['rows'])}")
    return 0


CASES = [
    ("clean", 0, "clean", None),
    ("target_needle_missing", 1, "failed", "target_line"),
    ("source_line_moved", 1, "failed", "source_line"),
    ("target_file_missing", 2, "could_not_evaluate", None),
    ("table_malformed", 2, "table_malformed", None),
    ("locus_row_absent", 1, "failed", "source_locus"),
    ("external_moved", 2, "could_not_evaluate", "external_moved"),
    ("external_ok", 0, "clean", None),
]


def selftest():
    with tempfile.TemporaryDirectory(prefix="citation-drift-selftest-") as tmp:
        fixture = Path(tmp)
        source = "# doc\nsee `target.md:1` for the pinned fact\ntail\n"
        target = "the pinned fact\n"
        pin = {"rows": [{"id": "r1", "kind": "pin", "source": "source.md",
                         "source_line": 2, "source_needle": "target.md:1",
                         "target": "target.md", "target_line": 1,
                         "target_needle": "pinned fact"}]}
        locus = {"rows": [{"id": "r2", "kind": "locus", "source": "source.md",
                           "source_needle": "target.md:1",
                           "target": "target.md", "target_line": 1,
                           "target_needle": "pinned fact"}]}
        external = {"rows": [{"id": "r3", "kind": "pin", "source": "source.md",
                              "source_line": 2, "source_needle": "target.md:1",
                              "target": "target.md", "target_line": 1,
                              "target_needle": "pinned fact",
                              "external": {"path": str(fixture / "host.conf"),
                                           "sha256": hashlib.sha256(
                                               target.encode()).hexdigest()}}]}
        (fixture / "source.md").write_text(source)
        (fixture / "target.md").write_text(target)
        (fixture / "host.conf").write_text(target)
        (fixture / "pin.json").write_text(json.dumps(pin))
        (fixture / "locus.json").write_text(json.dumps(locus))
        (fixture / "external.json").write_text(json.dumps(external))

        def run(table_path):
            command = [sys.executable, str(Path(__file__).resolve()),
                       "--root", str(fixture), "--table", str(table_path)]
            return subprocess.run(command, capture_output=True, text=True)

        results = []
        for case, code, marker, detail in CASES:
            (fixture / "source.md").write_text(source)
            (fixture / "target.md").write_text(target)
            (fixture / "host.conf").write_text(target)
            table_path = fixture / "pin.json"
            if case == "target_needle_missing":
                (fixture / "target.md").write_text("a different fact\n")
            elif case == "source_line_moved":
                (fixture / "source.md").write_text(
                    "# doc\nmoved\nsee `target.md:1` for the pinned fact\n")
            elif case == "target_file_missing":
                (fixture / "target.md").unlink()
            elif case == "table_malformed":
                table_path = fixture / "broken.json"
                table_path.write_text("{not json")
            elif case == "locus_row_absent":
                (fixture / "source.md").write_text("# doc\nno citation here\n")
                table_path = fixture / "locus.json"
            elif case == "clean":
                table_path = fixture / "locus.json"
            elif case == "external_moved":
                (fixture / "host.conf").write_text("operator rewrote me\n")
                table_path = fixture / "external.json"
            elif case == "external_ok":
                table_path = fixture / "external.json"
            run_result = run(table_path)
            output = (run_result.stdout + run_result.stderr).lower()
            passed = run_result.returncode == code and marker.lower() in output
            if detail is not None:
                passed = passed and detail.lower() in output
            print(f"SELFTEST_CASE: {'CLEAN' if passed else 'FAILED'} case={case} "
                  f"exit={run_result.returncode} expected={code}")
            results.append(passed)
        if not all(results):
            print(f"CITATION-DRIFT-SELFTEST: FAILED cases={len(results)}")
            return 1
        print(f"CITATION-DRIFT-SELFTEST: CLEAN cases={len(results)} exit=0")
        return 0


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--root", type=Path, default=HERE.parents[2])
    parser.add_argument("--table", type=Path, default=HERE / "pointers.json")
    parser.add_argument("--selftest", action="store_true")
    args = parser.parse_args()
    if args.selftest:
        return selftest()
    try:
        table = load_table(args.table)
        return report(args.root, table)
    except Finding as error:
        print(f"CITATION-DRIFT: COULD_NOT_EVALUATE finding={error.name}")
        return error.code
    except (OSError, ValueError, KeyError, TypeError):
        print("CITATION-DRIFT: COULD_NOT_EVALUATE "
              "finding=unreadable_or_invalid_input")
        return 2


if __name__ == "__main__":
    sys.exit(main())
```
