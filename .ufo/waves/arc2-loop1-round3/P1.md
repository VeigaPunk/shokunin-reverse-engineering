        toks = [int(m.group(1)) for m in RD_PTR.finditer(text)]
        if not toks:
            return {"file": rel, "line": line, "kind": kind, "ok": False,
                    "disp": "PARSE-FAILURE", "why": "no README token"}
        n = toks[-1]
        ok = 1 <= n <= len(rd) and "Runtime-obs" in rd[n - 1]
        anchor = re.compile(r"Runtime-obs")
    elif kind == "cfgptr":
        toks = [(int(m.group(1)), int(m.group(2)) if m.group(2) else None)
                for m in CFG_PTR.finditer(text)]
        toks += [(int(m.group(1)), int(m.group(2))) for m in CFG_SED.finditer(text)]
        if not toks:
            return {"file": rel, "line": line, "kind": kind, "ok": False,
                    "disp": "PARSE-FAILURE", "why": "no config pointer"}
        ok = all(cfg_ok(cfg, a, b) for (a, b) in toks)
        anchor = re.compile(re.escape(ANCHOR))
    else:
        ok, why = pin_ok(text, pins)
        anchor = re.compile(re.escape(ANCHOR))
    rec = {"file": rel, "line": line, "kind": kind, "ok": ok}
    if ok:
        rec["disp"] = "live"
    else:
        d = discharged(errata_sections(lines), line, anchor)
        rec["disp"] = "discharged" if d else "DEAD-UNDISCHARGED"
        rec["discharged_by"] = d
    return rec


def evaluate(root, config):
    cfg = readlines(config)
    rd = readlines(os.path.join(root, "corpus/README.md"))
    pins = live_pins(config)
    gated = []
    for (rel, line, kind, tier) in GATED:
        rec = eval_locus(root, rel, line, kind, cfg, rd, pins)
        rec["tier"] = tier
        gated.append(rec)
    tierb = []
    for (rel, line, kind, tier, token) in TIER_B:
        rec = eval_locus(root, rel, line, kind, cfg, rd, pins)
        rec["tier"] = tier
        if tier in ("attributed", "dated-section") and token:
            path = os.path.join(root, rel)
            if os.path.exists(path):
                lines = readlines(path)
                if tier == "dated-section":
                    j = line - 1
                    while j > 0 and not lines[j].startswith("## "):
                        j -= 1
                    rec["attribution_present"] = token in lines[j]
                else:
                    j, k = line - 1, line
                    while j > 0 and lines[j - 1].strip() and not lines[j - 1].startswith("- "):
                        j -= 1
                    while k < len(lines) and lines[k].strip():
                        k += 1
                    rec["attribution_present"] = token in "\n".join(lines[j:k])
        tierb.append(rec)
    dead = [r for r in gated if r["disp"] == "DEAD-UNDISCHARGED"]
    pf = [r for r in gated if r["disp"] == "PARSE-FAILURE"]
    return gated, tierb, dead, pf, pins


def selftest(root, config):
    tmp = tempfile.mkdtemp(prefix="arc2-l3r3-P1-selftest-")
    files = ["SCORECARD.md", "corpus/README.md",
             "corpus/knowledge/2026-10-13-mcp-registrations.md",
             "corpus/knowledge/2026-10-13-r4-16-frontier-rescan.md"]
    OLD = "`:157` pointer (at `:160` in HEAD, per seat C7) is recorded as named debt,"
    NEW = "`:157` pointer (at `:160` in HEAD, per seat C7; now `:177`, discharged at `7125dd4`) is recorded as named debt,"

    def seed(name):
        d = os.path.join(tmp, name)
        for f in files:
            os.makedirs(os.path.dirname(os.path.join(d, f)) or d, exist_ok=True)
            shutil.copy(os.path.join(root, f), os.path.join(d, f))
        return d

    def apply(d, sc=True, mcp=True, r4=True, drop_anchor=False, drop_receipt=False,
              drop_naming=None):
        if sc:
            p = os.path.join(d, "SCORECARD.md")
            t = open(p).read()
            assert OLD in t, "selftest: SCORECARD old line not found"
            open(p, "w").write(t.replace(OLD, NEW, 1))
        for flag, rel, err in ((mcp, "corpus/knowledge/2026-10-13-mcp-registrations.md", MCP_ERRATA),
                               (r4, "corpus/knowledge/2026-10-13-r4-16-frontier-rescan.md", R4_ERRATA)):
            if not flag:
                continue
            p = os.path.join(d, rel)
            t = open(p).read() + err
            if drop_anchor:
                t = t.replace(ANCHOR, "[stanza]")
            if drop_receipt:
                t = t.replace(RECEIPT_TOKEN, "<receipt>")
            if drop_naming:
                t = t.replace("`:%d`" % drop_naming, "`:<%d>`" % drop_naming)
            open(p, "w").write(t)

    cases = [("as-is", seed("as-is"), 1, None),
             ("after", seed("after"), 0, {}),
             ("neg-anchor", seed("na"), 1, {"drop_anchor": True}),
             ("neg-receipt", seed("nr"), 1, {"drop_receipt": True}),
             ("neg-naming", seed("nn"), 1, {"drop_naming": 88}),
             ("nocfg", seed("as-is"), 2, None)]
    rc_all = 0
    for (name, d, want, kw) in cases:
        cfg = config if name != "nocfg" else os.path.join(tmp, "missing.toml")
        if kw is not None:
            apply(d, **kw)
        if not os.path.exists(cfg):
            got = 2
        else:
            gated, tierb, dead, pf, pins = evaluate(d, cfg)
            got = 0 if (not dead and not pf) else 1
        ok = got == want
        rc_all |= 0 if ok else 1
        print("selftest %-11s -> exit %d (want %d) %s%s" % (
            name, got, want, "OK" if ok else "FAIL",
            ("  dead=%d" % len(dead)) if got == 1 else ""))
    print("selftest fixtures: %s" % tmp)
    return rc_all
