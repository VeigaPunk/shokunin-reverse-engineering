def paragraph_bounds(lines, idx):
    a = idx
    while a > 0 and lines[a - 1].strip():
        a -= 1
    b = idx
    while b + 1 < len(lines) and lines[b + 1].strip():
        b += 1
    return a, b


def sentence_span(ptext, off):
    lo = 0
    for m in SENT_SPLIT.finditer(ptext):
        if m.end() <= off:
            lo = m.end()
        elif m.start() > off:
            return lo, m.start()
    return lo, len(ptext)


def sentence_text(lines, li, start, end):
    a, b = paragraph_bounds(lines, li)
    ptext = " ".join(lines[k].strip() for k in range(a, b + 1))
    lead = len(lines[li]) - len(lines[li].lstrip())
    cur, off = 0, None
    for k in range(a, b + 1):
        s = lines[k].strip()
        if k == li:
            off = cur + max(0, start - lead)
        cur += len(s) + 1
    if off is None:
        return ""
    lo, hi = sentence_span(ptext, off)
    return ptext[lo:hi]

    if in_region:

            add("rd", m, int(m.group(1)), int(m.group(2) or m.group(1)),
                m.group(0), "readme")
        for m in BARE.finditer(line):
            add("bare-rd", m, int(m.group(1)), int(m.group(1)), m.group(1), "readme")
    return insts


def range_status(rows, expect, name):
    if not rows:
        return "dead", "cited range beyond EOF (%s has %d lines)" % (name, len(rows))
    if any(re.search(expect, s) for s in rows):
        return "live", "cited range carries the expected content"
    return "dead", "cited line lacks cited content: %r" % redact(rows[0])


def resolve(inst, cfg_lines, rd_head, rd_base, line):
    """Return (status, why, resolved_at, head_status)."""
    if inst["target"] == "config":
        ov = EXPECT_OVERRIDE.get((inst["file"], inst["token"]))
        if ov:
            seg = cfg_lines[inst["first"] - 1:inst["last"]] \
                if 1 <= inst["first"] <= inst["last"] <= len(cfg_lines) else []
            st, why = range_status(seg, ov, "config")
            return st, why, "config-live", st
        lo, hi = BLOCK[0], BLOCK[1]
        if 1 <= inst["first"] <= len(cfg_lines) and inst["first"] <= hi \
                and inst["last"] >= lo:
            return "live", "cited range intersects the live rea-stanza block %d-%d" % (lo, hi), \
                "config-live", "live"
        seg = cfg_lines[inst["first"] - 1:inst["first"]] \
            if 1 <= inst["first"] <= len(cfg_lines) else []
        return "dead", "cited line lacks cited content: %r" % (
            redact(seg[0]) if seg else "<beyond EOF>"), "config-live", "dead"
    if inst["kind"] == "rd":
        seg = rd_head[inst["first"] - 1:inst["last"]] \
            if 1 <= inst["first"] <= inst["last"] <= len(rd_head) else []
        st, why = range_status(seg, README_EXPECT, "corpus/README.md")
        return st, why, "head", st
    win = line[max(0, inst["start"] - 15):inst["end"] + 15]
    explicit = bool(EXPLICIT_CURRENT.search(win))
    hseg = rd_head[inst["first"] - 1:inst["first"]] \
        if 1 <= inst["first"] <= len(rd_head) else []
    bseg = rd_base[inst["first"] - 1:inst["first"]] \
        if rd_base and 1 <= inst["first"] <= len(rd_base) else []
    hst, _ = range_status(hseg, README_EXPECT, "corpus/README.md")
    if explicit:
        return hst, "explicit-current token resolved at REV", "head-explicit", hst
    if rd_base is None:
        return hst, "no baseline available; resolved at REV", "head", hst
    bst, bwhy = range_status(bseg, README_EXPECT, "baseline corpus/README.md")
    return bst, "baseline convention: " + bwhy, "baseline", hst


def class_discharged_at(lines):
    """First line of a paragraph that universally declares the file's
    pointers dead (marker + universal quantifier in one paragraph)."""
    for i, l in enumerate(lines):
        if UNIVERSAL.search(l):
            a, b = paragraph_bounds(lines, i)
            if any(REGION_MARKER.search(lines[k]) for k in range(a, b + 1)):
                return a + 1
    return None


def token_discharged_at(lines, locus, tokens):
    """First line (> locus) naming one of the locus's cited token VALUES
    (`:1284`, `:1285-1292`, ...) inside a marker-carrying paragraph."""
    if not tokens:
        return None
    pats = [re.compile(r":%s(?!\d)" % re.escape(tok)) for tok in tokens]
    for i in range(locus, len(lines)):
        if any(p.search(lines[i]) for p in pats):
            a, b = paragraph_bounds(lines, i)
            if any(REGION_MARKER.search(lines[k]) for k in range(a, b + 1)):
                return i + 1
    return None


def discharged_at(lines, locus):
    for i in range(locus, len(lines)):
        if re.search(r"(?<![0-9]):%d(?!\d)" % locus, lines[i]):
            a, b = paragraph_bounds(lines, i)
            if any(REGION_MARKER.search(lines[k]) for k in range(a, b + 1)):
                return i + 1
    return None
