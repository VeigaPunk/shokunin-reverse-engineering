## Evidence

The round-1 wave report describes a 64-seat wave and locates the seat reports under `.ufo/waves/arc2-loop1-round1/` (`.ufo/reports/arc2-loop1-round1/kaidoku.md:3-5`). I used its direct `.md` report filenames as the comparison roster. The following read-only command counts direct `.md` files and compares the ID sets; numeric ranges in its output are inclusive.

Command (run from the repository root at HEAD `0e8c96b48a8823e8c6964081cfc04e21f6adcf56`):

```bash
python3 - <<'PY'
from pathlib import Path
import re
root = Path('.ufo/waves')
def reports(name):
    return {p.stem for p in (root / name).glob('*.md') if p.is_file()}
def compact(ids):
    groups = {}
    for seat in ids:
        m = re.fullmatch(r'([A-Z]+)([0-9]+)', seat)
        if not m:
            groups.setdefault('?', []).append(seat)
        else:
            groups.setdefault(m.group(1), []).append(int(m.group(2)))
    out = []
    for prefix in sorted(groups):
        values = sorted(groups[prefix])
        spans = []
        start = end = values[0] if values else None
        for value in values[1:]:
            if value == end + 1:
                end = value
            else:
                spans.append((start, end))
                start = end = value
        if values:
            spans.append((start, end))
        out.extend(f'{prefix}{a}' if a == b else f'{prefix}{a}-{prefix}{b}' for a, b in spans)
    return ','.join(out)
roster = reports('arc2-loop1-round1')
print(f'arc2-loop1-round1: roster_reports={len(roster)} roster={compact(roster)}')
rounds = [f'arc2-loop1-round{i}' for i in (2, 3, 4)]
sets = []
for name in rounds:
    found = reports(name)
    sets.append(found)
    print(f'{name}: reports={len(found)} present={compact(found)} missing={compact(roster-found)} extra={compact(found-roster) or "-"}')
print(f'missing_sets_repeat={all(roster-s == roster-sets[0] for s in sets[1:])}')
print(f'report_sets_identical={all(s == sets[0] for s in sets[1:])}')
PY
```
