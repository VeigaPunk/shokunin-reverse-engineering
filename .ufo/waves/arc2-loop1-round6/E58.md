def command_a_from_text(rev, readme, skill, knowledge):
    files = {os.path.splitext(os.path.basename(p))[0] for p in knowledge if p.endswith(".md")}

    mapped = set(re.findall(r"2026-10-\d{2}-[a-z0-9-]+", section))
    craft = skill.split("## Queue-drain craft ", 1)[1].split("\n## ", 1)[0]
    rules = re.split(r"(?m)^- \*\*", craft)[1:]
    pattern = r"corpus/knowledge/([a-z0-9-]+\.md)(?::(\d+)(?:-(\d+))?)?"
    resolved = 0
    for rule in rules:
        refs = re.findall(pattern, rule)
        valid = bool(refs)
        for name, first, last in refs:
            fp = os.path.join(ROOT, "corpus/knowledge", name)
            if not os.path.isfile(fp):
                valid = False
                continue
            count = len(open(fp, encoding="utf-8").read().splitlines())
            start, end = int(first or 1), int(last or first or 1)
            valid = valid and 1 <= start <= end <= count
        resolved += valid
    inter = files & mapped
    out = {
        "mapped": len(inter),
        "knowledge_files": len(files),
        "missing": sorted(files - mapped),
        "dangling": sorted(mapped - files),
        "citation_resolved_rules": resolved,
        "rule_blocks": len(rules),
        "coverage_score": round(10 * len(inter) / len(files), 2),
        "depth_score": round(10 * resolved / len(rules), 2),
    }
    ok = (out["missing"] == [] and out["dangling"] == []
          and out["knowledge_files"] in (79, 80) and out["mapped"] == out["knowledge_files"]
          and out["rule_blocks"] == 11 and out["citation_resolved_rules"] == 11
          and out["coverage_score"] == 10.0 and out["depth_score"] == 10.0)
    new = files - head_files
    if out["knowledge_files"] == 79:
        ok = ok and new == set() and mapped == head_mapped
    else:
        ok = ok and len(new) == 1 and new <= mapped
    note(ok, "G1 command-A worktree", json.dumps(out, sort_keys=True) + " new=" + str(sorted(new)))
    map_lines = len(section.splitlines())
    note(map_lines == HEAD_MAP_LINES, "G1 coverage-map line-neutral",
         "map_lines=%d expected=%d" % (map_lines, HEAD_MAP_LINES))
    note(LESSON_TOKEN not in section, "G1 lesson-token not in coverage map", LESSON_TOKEN)
    print("NOTE G1: knowledge_files=%d is the repo shape only. 80 = HELD map-token; "
          "79 = honest-negative or C1 not landed. This script does not read "
          "arc2-r6-pe. An in-flight service at close means C1 is not verified."
          % out["knowledge_files"])
    return out, section, mapped, files

def g_rev(rev):
    p = run(["git", "rev-parse", "--verify", rev + "^{commit}"])
    note(p.returncode == 0, "G3 --rev parses", p.stdout.strip())
    if p.returncode != 0:
        return
    full = p.stdout.strip()
    head = run(["git", "rev-parse", "HEAD"]).stdout.strip()
    note(head == full, "G0 HEAD==--rev", "HEAD %s rev %s" % (head, full))
    dirty = run(["git", "diff", "--exit-code", "--"] + GRADED)
    note(dirty.returncode == 0, "G0 worktree==HEAD on graded paths", "rc=%d" % dirty.returncode)
    out, _, _, _ = command_a_rev(full)
    ok = (out["missing"] == [] and out["dangling"] == []
          and out["knowledge_files"] in (79, 80) and out["mapped"] == out["knowledge_files"]
          and out["rule_blocks"] == 11 and out["citation_resolved_rules"] == 11
          and out["coverage_score"] == 10.0 and out["depth_score"] == 10.0)
    note(ok, "G1 command-A git-show --rev", json.dumps(out, sort_keys=True))
    rc, text = probe(full, strict=False)
    note(rc == 0 and verdict_line(text) == "CITATION_DRIFT_VERDICT: reported findings=0 exit=0",
         "G3 registered probe --rev", "rc=%d %s" % (rc, verdict_line(text)))
    rc, text = probe(full, strict=True)
    note(rc == 0 and verdict_line(text) == "CITATION_DRIFT_VERDICT: clean findings=0 exit=0",
         "G3 registered probe --rev --strict", "rc=%d %s" % (rc, verdict_line(text)))
