## Verbatim-landable closeout command

import hashlib, re
from pathlib import Path
wave = Path(".ufo/waves/arc2-loop1-round6")
e26 = (wave / "E26.md").read_text().splitlines()
# fence opened at line 38 in the S3-verified extract; re-find the last text fence

blocks = list(re.finditer(r"```text\n(.*?)```", text, re.S))
body = blocks[-1].group(1)
if not body.endswith("\n"):
    body += "\n"
digest = hashlib.sha256(body.encode()).hexdigest()
pin = "4bde2867bd859efc1173aa6685b1da014b0f48b46a5e3fc3aa81c87e31ee8463"
print(f"fence_sha256={digest} pin_match={digest == pin} bytes={len(body.encode())}")
if digest != pin:
    raise SystemExit(2)
roster = []
for ln in body.splitlines():
    if not ln.strip() or ln.startswith("n\t"):
        continue
    roster.append(ln.split("\t")[1])
if len(roster) != 128 or len(set(roster)) != 128:
    raise SystemExit(f"roster_len={len(roster)} unique={len(set(roster))}")
disk = {p.stem: p.stat().st_size for p in wave.glob("*.md")}
valid = {s for s in roster if disk.get(s, 0) > 0}
missing = [s for s in roster if s not in valid]
unexpected = sorted(set(disk) - set(roster))
