```python
def rev_ok(root, rev):
    """True iff --rev names one commit. A missing or non-commit rev is
    could-not-evaluate, not a set of absent files. Never raises."""
    proc = subprocess.run(
        ["git", "-C", root, "rev-parse", "--verify", "--quiet",
         "%s^{commit}" % rev],
        capture_output=True)
    return proc.returncode == 0
```
