```
mkdir x-vendor && sasquatch-v4be -f -d x-vendor avm-be-member.sqfs
```

## Oracle

```
for spec in "vendor-be avm avm-be-member.sqfs" "vendor-le le avm-le-member.sqfs" \
            "r11 avm avm-be.squashfs" "k1 be k1.sqsh" ... "k6 be k6.sqsh"; do
  set -- $spec
  python3 -I -B $A/sqfs-list.py --layout $2 --files out/$1.files.tsv \
      --nonregular out/$1.nonregular.tsv --inspect $3 > out/$1.inspect.json
  python3 -I -B $A/sqfs-list.py --layout auto --files out/$1.auto.files.tsv \
      --inspect $3 > out/$1.auto.json
  cmp out/$1.files.tsv out/$1.auto.files.tsv        # identical on all nine
done
```
