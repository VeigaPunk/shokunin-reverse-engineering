def same_rel(candidate, rel):
    return bool(candidate) and bool(rel) and (candidate == rel or candidate.endswith("/" + rel))
def nonregular_published(files, nonreg):
    hits = []
    for fp, _, _ in files:
        for rel, _kind in nonreg:
            if same_rel(fp, rel): hits.append(fp)
    return hits
