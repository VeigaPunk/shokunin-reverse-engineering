```python
    for os_, oc, ns_, nc in hs:
        if os_ == 0 or oc == 0:                       # pure insertion/append
            if os_ == oldlines and c["append"]:
                continue
            if os_ in c["inserts"]:
                continue
            bad.append(f"insert/append hunk not allowed @@ -{os_},{oc} +{ns_},{nc}")
            continue
        lo, hi = os_, os_ + oc - 1
        if not any(lo >= a and hi <= z for a, z in c["allow"]):
            bad.append(f"hunk @@ -{os_},{oc} outside repair envelope {c['allow']}")
    if dele > c["del_budget"]:
        bad.append(f"deletions {dele} > budget {c['del_budget']}")
    new = run("git", "show", f":{path}").stdout
    for frag in c["protect"]:
        if frag not in new:
            bad.append(f"{HIST}: {frag[:56]!r}")
    for label, rx in c.get("require", []):
        if not re.search(rx, new, re.M):
            bad.append(f"qualification-not-erasure guard: {label}")
    print(f"{'FAIL' if bad else 'OK  '} {path}: +{add} -{dele} hunks={len(hs)} "
          + " ".join("@@ -%d,%d +%d,%d" % h for h in hs))
    if bad:
        rc = 1
        for x in bad:
            print(f"       {x}")

if EXPECT is not None:
    got = set(staged)
    if got != EXPECT:
        print(f"FAIL staged-set: got {sorted(got)} expected {sorted(EXPECT)}")
        rc = 1
    else:
        print(f"OK   staged-set exact: {sorted(got)}")
elif staged:
    print(f"NOTE staged paths outside contract: {sorted(staged)}")
sys.exit(rc)
```
