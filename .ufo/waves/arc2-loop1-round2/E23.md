```python
#!/usr/bin/env python3
"""C2 counting instrument (draft, seat E23, wave arc2-loop1-round2).

Unqualified-safety-label census over EDITABLE CORPUS FILES at a revision.
Reads immutable Git objects (never the moving worktree). Run from repo root,
after the edits under test are committed.

Usage: python3 c2-census.py <REV> [FROZEN_REF]

counted scope : every tracked corpus/**/*.md at REV. LOG.md / SCORECARD.md are
                NOT corpus files -> outside the counted scope by construction;
                their needle lines are printed under frozen_files, and (when
                FROZEN_REF is given) the frozen regions are checked against it:
                LOG.md must be prefix-identical (append-only history) and the
                SCORECARD.md "Historical scope bleed" bullet (:77-80, holds the
                frozen :79) byte-identical.
needle        : /benign|malicious/ CASE-INSENSITIVE over the whole file.
note          : a line containing the literal marker '**Safety-verdict scope'
                plus its continuation lines.
                strict extent = marker line + INDENTED non-empty continuation
                lines (this extent is what exempts a needle line as the note's
                own wording).
                loose extent  = marker line + all following non-empty lines
                until a block-start line (`- `/`* `/`#`/`|`/`>`/fence/`N. `);
                bare ':N' / ':N-M' naming tokens are collected from here, so a
                note may name loci in a plain paragraph as well as in indented
                continuations.
qualification : a needle line is qualified iff it carries the marker, lies
                inside a note's strict extent, or is named by a bare ':N' /
                ':N-M' token inside a note's loose extent. Post-note text is
                always scanned: no escape hatch.
exclusions    : declared non-locus / debt-naming loci, live-asserted to still
                carry the needle at REV; a stale exclusion FAILS the run.
gate          : exit 0 iff n_unqualified == 0 (STRICT) and no marker_missing
                and no stale_exclusions and no frozen_guard failure. n==0 is
                never trusted on its own -- marker_missing and untrusted_zero
                are reported before it can be believed. The loose count is a
                diagnostic, not the gate: strict!=0 with loose==0 means the
                note's naming tokens sit on unindented continuation lines
                (reword the note, do not loosen the gate).
residuals     : (1) the needle is not a full safety vocabulary; lines matching
                WIDE (safe|trust|compromis|...) are printed under wide_review
                for human inspection and are NOT gated. (2) a needle line added
                INSIDE a note's strict extent is exempt by construction -- a
                note necessarily quotes the labels it qualifies; keep notes
                dated, append-only, and quoting only historical wording.
"""
import hashlib
import json
import re
import subprocess
import sys

if len(sys.argv) < 2:
    sys.exit(__doc__)
rev = sys.argv[1]
frozen_ref = sys.argv[2] if len(sys.argv) > 2 else None

NEEDLE = re.compile(r'benign|malicious', re.I)
WIDE = re.compile(r'safe|trust|compromis|legit|endors|tamper|suspicious|uncompromised|harmless', re.I)
MARKER = '**Safety-verdict scope'
BARE = re.compile(r'(?<![A-Za-z0-9_./-]):(\d+)(?:-(\d+))?')
BLOCK_START = re.compile(r'(- |\* |#|\||>|```|\d+\. )')

EXCLUSIONS = {
    'corpus/knowledge/2026-10-13-crash-evidence.md:36':
        'non-safety-sense: journal noise characterization (CR5.md:97)',
    'corpus/knowledge/2026-10-14-r11-01-queue-ops.md:383':
        'non-safety-sense: telemetry size characterization (CR5.md:98)',
    'corpus/lessons/2026-10-15-round-1.md:150':
        'debt-naming: inside its own correction-debt sentence (CR5.md:91)',
}

NARROW_SCOPE = ['corpus/knowledge/2026-10-13-r3-jq-provenance.md', 'corpus/README.md']
FROZEN_FILES = ['LOG.md', 'SCORECARD.md']
SCORECARD_FROZEN_MARKER = 'Historical scope bleed'
SCORECARD_FROZEN_SPAN = 4


def get(path, ref=None):
    return subprocess.check_output(
        ['git', 'show', f'{ref or rev}:{path}'], text=True, stderr=subprocess.DEVNULL)


def sha(text):
    return hashlib.sha256(text.encode()).hexdigest()


def note_blocks(lines):
    """(marker_lines, strict_ranges, loose_ranges, named_set, named_ranges)."""
    marker, strict, loose, named, ranges = [], [], [], set(), []
    for i, l in enumerate(lines):
        if MARKER not in l:
            continue
        marker.append(i + 1)
        j = i + 1
        while j < len(lines) and lines[j][:1] in (' ', '\t') and lines[j].strip():
            j += 1
        strict.append((i + 1, j))
        k = i + 1
        while k < len(lines) and lines[k].strip() and not BLOCK_START.match(lines[k]):
            k += 1
        loose.append((i + 1, k))
        for l2 in lines[i:k]:
            for a, b in BARE.findall(l2):
                named.update(range(int(a), int(b or a) + 1))
                ranges.append(f':{a}-{b}' if b else f':{a}')
    return marker, strict, loose, named, ranges


def census(path):
    lines = get(path).splitlines()
    needle = [i + 1 for i, l in enumerate(lines) if NEEDLE.search(l)]
    marker, strict, loose, named, ranges = note_blocks(lines)
    in_strict = lambda x: any(a <= x <= b for a, b in strict)
    in_loose = lambda x: any(a <= x <= b for a, b in loose)
    excluded = [int(n) for p, n in (k.rsplit(':', 1) for k in EXCLUSIONS)
                if p == path and int(n) in needle]
    unq = [x for x in needle if x not in excluded and not in_strict(x)
           and x not in named and MARKER not in lines[x - 1]]
    unq_loose = [x for x in needle if x not in excluded and not in_loose(x)
                 and x not in named and MARKER not in lines[x - 1]]
    untrusted = []
    if needle and not unq and not marker and not excluded:
        untrusted.append('n=0 with neither a marker note nor an exclusion')
    wide = [x for x in range(1, len(lines) + 1)
            if WIDE.search(lines[x - 1]) and x not in needle
            and not in_strict(x) and MARKER not in lines[x - 1]]
    return {'file': path, 'needle': needle, 'marker_lines': marker,
            'strict_blocks': strict, 'loose_blocks': loose,
            'named_ranges': ranges, 'excluded': sorted(excluded),
            'unqualified': unq, 'unqualified_loose': unq_loose,
            'untrusted_zero': untrusted, 'wide_review': wide}


scope = sorted(p for p in subprocess.check_output(
    ['git', 'ls-tree', '-r', '--name-only', rev, '--', 'corpus'],
    text=True).splitlines() if p.endswith('.md'))
rows = [census(p) for p in scope]
n = sum(len(r['unqualified']) for r in rows)
n_loose = sum(len(r['unqualified_loose']) for r in rows)
marker_missing = [r['file'] for r in rows if r['unqualified'] and not r['marker_lines']]
untrusted = [{'file': r['file'], 'why': r['untrusted_zero']} for r in rows if r['untrusted_zero']]

stale = []
for key in EXCLUSIONS:
    p, ln = key.rsplit(':', 1)
    if p not in scope:
        stale.append({'exclusion': key, 'reason': 'file absent at REV'})
        continue
    lines = get(p).splitlines()
    if not (1 <= int(ln) <= len(lines)) or not NEEDLE.search(lines[int(ln) - 1]):
        stale.append({'exclusion': key, 'reason': 'line no longer carries the needle'})

frozen = {}
for p in FROZEN_FILES:
    try:
        frozen[p] = [i + 1 for i, l in enumerate(get(p).splitlines()) if NEEDLE.search(l)]
    except subprocess.CalledProcessError:
        frozen[p] = 'absent at REV'

guard = []
try:
    after_log = get('LOG.md')
    entry = {'name': 'LOG.md prefix (append-only)', 'ref': frozen_ref, 'ok': None,
             'sha256_at_rev': sha(after_log)}
    if frozen_ref:
        before_log = get('LOG.md', frozen_ref)
        entry['ok'] = after_log.startswith(before_log)
        entry['sha256_at_ref'] = sha(before_log)
    guard.append(entry)
except subprocess.CalledProcessError:
    guard.append({'name': 'LOG.md absent at REV', 'ok': False})
try:
    sc = get('SCORECARD.md').splitlines()
    idx = [i for i, l in enumerate(sc) if SCORECARD_FROZEN_MARKER in l]
    if len(idx) == 1:
        i = idx[0]
        bullet = '\n'.join(sc[i:i + SCORECARD_FROZEN_SPAN]) + '\n'
        entry = {'name': f'SCORECARD.md frozen bullet (:{i + 1}-{i + SCORECARD_FROZEN_SPAN})',
                 'sha256_at_rev': sha(bullet), 'ok': None}
        if frozen_ref:
            scb = get('SCORECARD.md', frozen_ref).splitlines()
            jdx = [j for j, l in enumerate(scb) if SCORECARD_FROZEN_MARKER in l]
            if len(jdx) == 1:
                j = jdx[0]
                b2 = '\n'.join(scb[j:j + SCORECARD_FROZEN_SPAN]) + '\n'
                entry['ok'] = sha(b2) == sha(bullet)
                entry['sha256_at_ref'] = sha(b2)
            else:
                entry['ok'] = False
                entry['note'] = f'marker found {len(jdx)}x at ref'
        guard.append(entry)
    else:
        guard.append({'name': f'SCORECARD.md {SCORECARD_FROZEN_MARKER} occurrences',
                      'count': len(idx), 'ok': False})
except subprocess.CalledProcessError:
    guard.append({'name': 'SCORECARD.md absent at REV', 'ok': False})

pointer_probe = []
pointer_drift = []
try:
    sc = get('SCORECARD.md').splitlines()
    rd = get('corpus/README.md').splitlines()
    for i, l in enumerate(sc):
        for m in re.finditer(r'corpus/README\.md:(\d+)(?:-(\d+))?', l):
            a = int(m.group(1))
            target = rd[a - 1] if 1 <= a <= len(rd) else '<out of range>'
            pointer_probe.append({'sc_scorecard_line': i + 1, 'token': m.group(0),
                                  'readme_line': a, 'text': target[:80]})
    # C2's drifted-pointer measurement. The SCORECARD stale-summaries bullet
    # ("still lists already-held ... as frontier") carries two README pointers;
    # each must resolve to the content it describes.
    # Clause 1 (the C2 claim): the corpus/README.md:N token must land inside the
    # `## Frontier` section on the already-held bullet. ok is the measured bit.
    # Clause 2 (diagnostic): the bare :M token must land on the round-13
    # nit-count row. Reported, not gated.
    fstart = next((j + 1 for j, t in enumerate(rd) if t.startswith('## Frontier')), None)
    fend = next((j + 1 for j, t in enumerate(rd)
                 if fstart and j + 1 > fstart and t.startswith('## ')), len(rd) + 1)
    for i, l in enumerate(sc):
        if 'still lists already-held' not in l:
            continue
        m = re.search(r'corpus/README\.md:(\d+)', l)
        a = int(m.group(1)) if m else None
        ok = bool(a and fstart and fstart <= a < fend and 'frontier' in rd[a - 1].lower())
        clause = {'sc_line': i + 1, 'clause': 1, 'token': f'corpus/README.md:{a}',
                  'frontier_section': [fstart, fend - 1], 'ok': ok,
                  'resolves_to': rd[a - 1][:70] if a and 1 <= a <= len(rd) else '<out of range>'}
        pointer_drift.append(clause)
        m2 = re.search(r'`:(\d+)` says four r13', '\n'.join(sc[i:i + 3]))
        if m2:
            b = int(m2.group(1))
            tgt = rd[b - 1] if 1 <= b <= len(rd) else '<out of range>'
            nit_lines = [j + 1 for j, t in enumerate(rd) if '4 nits' in t or 'four r13' in t]
            pointer_drift.append({'sc_line': i + 1, 'clause': 2, 'token': f':{b}',
                                  'ok': '4 nits' in tgt or 'four r13' in tgt,
                                  'nit_count_lines': nit_lines,
                                  'resolves_to': tgt[:70]})
        break
except subprocess.CalledProcessError:
    pass

hints = []
for r in rows:
    if r['unqualified'] and r['marker_lines'] and not r['unqualified_loose']:
        hints.append({'file': r['file'], 'strict_only': r['unqualified'],
                      'hint': 'note present; naming tokens on unindented continuation lines'})

narrow = {r['file']: r['unqualified'] for r in rows if r['file'] in NARROW_SCOPE}
guard_ok = all(g.get('ok') is not False for g in guard)
out = {
    'revision': rev,
    'frozen_ref': frozen_ref,
    'counted_scope_files': len(scope),
    'n_unqualified': n,
    'n_unqualified_loose': n_loose,
    'unqualified': [{'file': r['file'], 'lines': r['unqualified']} for r in rows if r['unqualified']],
    'unqualified_loose': [{'file': r['file'], 'lines': r['unqualified_loose']} for r in rows if r['unqualified_loose']],
    'marker_missing': marker_missing,
    'untrusted_zero': untrusted,
    'hints': hints,
    'stale_exclusions': stale,
    'frozen_guard': guard,
    'frozen_files': frozen,
    'pointer_probe': pointer_probe,
    'pointer_drift': pointer_drift,
    'narrow_scope': narrow,
    'narrow_scope_n': sum(len(v) for v in narrow.values()),
    'wide_review': [{'file': r['file'], 'lines': r['wide_review']} for r in rows if r['wide_review']],
    'files': {r['file']: r for r in rows if r['needle']},
}
print(json.dumps(out, sort_keys=True, indent=1))
sys.exit(0 if (n == 0 and not marker_missing and not stale and not untrusted and guard_ok) else 1)
```
