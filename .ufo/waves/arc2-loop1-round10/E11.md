## 3. Verbatim-landable text

Instrument, landable to `.ufo/gates/convergence-census/convergence-census.py`
(reads wave dirs; exit 0 always; prints coverage + misattributions; `--selftest`
runs the mutation control):

```python
#!/usr/bin/env python3
"""Convergence census — per-wave, per-claim agreement across seat reports.

A claim is a canonical FINGERPRINT VALUE: a value that uniquely identifies one
artifact (byte size, procedure/segment/string count, evidence id, wall, peak).
Coverage alone is unfalsifiable; the conflict detector is mutation-validated:

    convergence-census.py . --selftest .ufo/waves/arc2-loop1-round7/E3.md

Expect 3/3 PASS. A census that has not been shown able to detect a conflict
reports nothing.
"""
import re, os, sys, json, collections, shutil, tempfile

CANON = {
 'bytes_big':('big','91,662,336',r'91[,.]?662[,.]?336'),
 'bytes_mid':('mid','17,694,720',r'17[,.]?694[,.]?720'),
 'procs_big':('big','75,218',r'75[,\s]?218'),
 'procs_mid':('mid','19,012',r'19[,\s]?012'),
 'procs_bisect':('bisect','6,337',r'6[,\s]?337'),
 'segments_big':('big','9',r'\b9 segments\b'),
 'segments_mid':('mid','11',r'\b11 segments\b'),
 'strings_big':('big','137,345',r'137[,\s]?345'),
 'strings_mid':('mid','4,382',r'\b4382 strings\b'),
 'evid_big':('big','ev_a4aa9b0d…',r'ev_a4aa9b0d[0-9a-f]*'),
 'evid_mid':('mid','ev_8b688ce7…',r'ev_8b688ce7[0-9a-f]*'),
 'evid_bisect':('bisect','ev_92dea347…',r'ev_92dea347[0-9a-f]*'),
 'wall_big':('big','1424.663s',r'1424(?:\.\d+)?'),
 'wall_mid':('mid','2479s',r'\b2479\b'),
 'wall_ctl':('ctl','4m42.945s',r'282\.945|4m42(?:\.\d+)?s'),
 'peak_big':('big','7.9G',r'7\.9\s*G'),
}
SUBJECT_RE = {
 'big':r'91\.6\s*MB|godot|windows_release_arm64',
 'mid':r'17\.7\s*MB|agent-browser',
 'bisect':r'2\.5\s*MB|decompile\.exe',
 'ctl':r'natural control|r8-ctl|arc2-r8-ctl',
}
EV_FULL = {
 'ev_a4aa9b0d':('big','ev_a4aa9b0d81b0a9e9cd93a5a93399a32fa1aa4abe2076bbf41c69ca6246cde608'),
 'ev_8b688ce7':('mid','ev_8b688ce7b54ef1d08823734fda2f062fd19dfa2ca9f4023c329cbae22f726986'),
 'ev_92dea347':('bisect','ev_92dea347d9f9ffc38b101047d0e8f147b8b72c1d96371ba9e9f135e613b296b9'),
}
STRUCT = [
 ('procedures', r'([\d][\d,]{3,})\s*(?:procedures|procs)\b'),
 ('segments',   r'(\d{1,2})\s*segments\b'),
 ('strings',    r'([\d][\d,]{2,})\s*strings\b'),
]
CANON_STRUCT = {
 'procedures': {'75218':'big','19012':'mid','6337':'bisect'},
 'segments':   {'9':'big','11':'mid','7':'bisect'},
 'strings':    {'137345':'big','4382':'mid','2716':'bisect'},
}

def scan(text):
    hits = {}
    for key, (sub, disp, pat) in CANON.items():
        best = 0
        for m in re.finditer(pat, text):
            best = max(best, 2 if (key.startswith('evid') and len(m.group(0)) == 67) else 1)
        if best:
            hits[key] = best
    mis = []
    for m in re.finditer(r'ev_[0-9a-f]{64}\b', text):
        v = m.group(0)
        for pref, (sub, canon) in EV_FULL.items():
            if v.startswith(pref) and v != canon:
                mis.append((sub, 'evid', v, v[:40] + '...'))
    for line in text.replace('\r', '').split('\n'):
        for s in re.split(r'(?<=[.;])\s+', line):
            subs = [n for n, p in SUBJECT_RE.items() if re.search(p, s, re.I)]
            if len(subs) != 1:
                continue
            for fam, pat in STRUCT:
                for m in re.finditer(pat, s, re.I):
                    v = m.group(1).replace(',', '')
                    if v not in CANON_STRUCT[fam]:
                        mis.append((subs[0], fam, v, s.strip()[:100]))
    return hits, mis

def census(wdir):
    seats = sorted(f for f in os.listdir(wdir) if f.endswith('.md'))
    hits = collections.defaultdict(list); mis = collections.defaultdict(list)
    for s in seats:
        t = open(os.path.join(wdir, s), encoding='utf-8', errors='replace').read()
        h, m = scan(t)
        for k, v in h.items():
            hits[k].append((s, v))
        for it in m:
            mis[s].append(it)
    return seats, hits, mis

def report(wdir, label, quiet=False):
    seats, hits, mis = census(wdir)
    print(f'### {label}  seats={len(seats)}')
    if not seats:
        print('  (no reports)'); print()
        return dict(wave=label, seats=0, slots=0, mis=0)
    lines = []
    for key, (sub, disp, pat) in CANON.items():
        rows = hits.get(key, [])
        if not rows:
            continue
        f = sum(1 for _, v in rows if v == 2); tr = sum(1 for _, v in rows if v == 1)
        extra = f' full={f} trunc={tr}' if key.startswith('evid') else ''
        lines.append(f'  {key:14s} {sub:6s} seats={len(rows):3d}{extra}')
    if not quiet:
        print('\n'.join(lines))
    nm = sum(len(v) for v in mis.values())
    print(f'  canonical slots covered: {len(lines)}   structural misattributions: {nm}')
    for s, items in list(mis.items())[:8]:
        for sub, fam, v, snip in items[:2]:
            print(f'    !! {s}: [{sub}/{fam}]={v}  "{snip}"')
    print()
    return dict(wave=label, seats=len(seats), slots=len(lines), mis=nm)

if __name__ == '__main__':
    root = sys.argv[1]
    args = sys.argv[2:]
    if args and args[0] == '--selftest':
        srcf = None
        for c in args[1:]:
            t = open(c, encoding='utf-8', errors='replace').read()
            if re.search(r'75[,\s]?218\s*procedures', t) and re.search(r'\b9 segments\b', t) \
               and re.search(r'ev_a4aa9b0d[0-9a-f]{8}', t):
                srcf = c; break
        if not srcf:
            print('SELFTEST SKIP: no candidate carries all three canonical patterns in scoped form')
        else:
            t = open(srcf, encoding='utf-8', errors='replace').read()
            muts = [('procedures 75,218->75,999', t.replace('75,218', '75,999').replace('75218', '75999')),
                    ('segments 9->8', re.sub(r'\b9 segments\b', '8 segments', t)),
                    ('evid hex tail flip', re.sub(r'(ev_a4aa9b0d)[0-9a-f]', r'\1f', t))]
            for name, mt in muts:
                tmp = tempfile.mkdtemp(prefix='selftest-')
                open(os.path.join(tmp, 'A.md'), 'w').write(t)
                open(os.path.join(tmp, 'B.md'), 'w').write(mt)
                seats, hits, mis = census(tmp)
                det = sum(len(v) for v in mis.values())
                evmis = [x for v in mis.values() for x in v if x[1] == 'evid']
                ok = (len(evmis) > 0) if name.startswith('evid') else (det > 0)
                print(f'SELFTEST [{name}] source={srcf} struct_mis={det} evid_mis={len(evmis)} '
                      f'-> {"PASS" if ok else "FAIL"}')
                shutil.rmtree(tmp)
    else:
        res = [report(os.path.join(root, d), d) for d in args]
        print(json.dumps(res))
```
