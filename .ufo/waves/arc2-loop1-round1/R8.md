## Block B — checker, land verbatim if wanted (demonstrated; exit 0 real / 1 marker / 2 usage)

```bash
#!/usr/bin/env bash
# R8 marker-vs-report checker. Usage: check-report.sh <repo> <report-path> [<expected-round-work-hash>]
# Exit 0 = real report (maybe defective); 1 = marker; 2 = usage error.
set -u
repo=$1; rpt=$2; work=${3:-}
cd "$repo" || exit 2
[ -f "$rpt" ] || { echo "P1 path-exists: FAIL"; exit 1; }
echo "P1 path-exists: OK"

# P2 SHONEN heading
if grep -qE '^#{1,6}[[:space:]].*SHONEN' "$rpt"; then echo "P2 shonen-heading: OK"; else echo "P2 shonen-heading: FAIL (contract: report without this heading does not satisfy PROTOCOL.md:148-150)"; fi

# P3 one mark line per box (SHONEN.md has 4 boxes)
marks=$(awk '/^#{1,6}[[:space:]].*SHONEN/{f=1;next} f&&/^- \[[xX~ ]\]/{n++} END{print n+0}' "$rpt")
echo "P3 shonen-marks: $marks (expect 4)"

# P4 resolvable commit hashes
hashes=$(grep -oE '\b[0-9a-f]{7,40}\b' "$rpt" | sort -u)
ok=0; total=0
for h in $hashes; do total=$((total+1)); if git cat-file -e "$h^{commit}" 2>/dev/null; then ok=$((ok+1)); fi; done
echo "P4 hashes-resolvable: $ok/$total"
if [ -n "$work" ]; then
  if git cat-file -e "$work^{commit}" 2>/dev/null; then
    if git merge-base --is-ancestor "$work" HEAD; then echo "P4b ancestor($work): OK"; else echo "P4b ancestor($work): FAIL (not an ancestor of HEAD)"; fi
  else echo "P4b ancestor($work): FAIL (hash not in repo)"; fi
fi

# P5 report committed under lane identity
c=$(git log -1 --format='%H %an <%ae>' -- "$rpt" 2>/dev/null)
if [ -n "$c" ]; then echo "P5 report-committed: OK [$c]"; else echo "P5 report-committed: FAIL (untracked - lands-last clause unsatisfied, PROTOCOL.md:122)"; fi

# P6 resolvable evidence pointers (backticked tokens containing / or .md)
n=0; okp=0
while IFS= read -r tok; do
  case "$tok" in */*|*.md) n=$((n+1)); if [ -e "${tok%%:*}" ] || [ -e "$tok" ]; then okp=$((okp+1)); fi;; esac
done < <(grep -oE '`[^`]+`' "$rpt" | tr -d '`')
echo "P6 pointers-resolvable: $okp/$n (heuristic; .md-with-no-slash tokens resolve in repo root)"

# P7 required sections present (templates/round-prompt.md:62-73)
sec=0
for s in 'Work performed' 'Landed changes' 'Before.*after' 'SHONEN'; do grep -qiE "^#{1,6}[[:space:]].*$s" "$rpt" && sec=$((sec+1)); done
echo "P7 sections(work/landed/before-after/shonen): $sec/4"

# Verdict
if [ "$ok" -eq 0 ] && [ "$okp" -eq 0 ]; then
  echo "VERDICT: MARKER (no resolvable commit, no resolvable pointer -> integrator records failed, PROTOCOL.md:184-191)"; exit 1
elif [ "$marks" -lt 4 ] || [ "$sec" -lt 4 ]; then
  echo "VERDICT: DEFECTIVE REAL REPORT (names real committed work but minimum-content defect; status ok, defect recorded)"; exit 0
else
  echo "VERDICT: REAL REPORT (contract-satisfying on this checklist)"; exit 0
fi
```
